{{- /* A figure inside a project page. {{< figure src="images/projects/foo.png" caption="What it shows" >}} Pass width= to cap it — a logo or a portrait-shaped panel should not run the full column. `src` is relative to ./assets — the source is resized to 1440px wide (2x the 720px a figure displays at) and converted to WebP, so full-resolution originals can be committed without shipping them. A path outside ./assets is used as-is. */ -}} {{- $src := .Get "src" -}} {{- $caption := .Get "caption" -}} {{- $alt := .Get "alt" | default $caption -}} {{- $width := .Get "width" -}} {{- $w := "" -}}{{- $h := "" -}} {{- with resources.Get $src -}} {{- $img := . -}} {{- if gt .Width 1440 }}{{ $img = .Resize "1440x" }}{{ end -}} {{- if ne .MediaType.SubType "svg" }}{{ $img = $img.Process "webp q82" }}{{ end -}} {{- $src = $img.RelPermalink -}} {{- $w = $img.Width -}}{{- $h = $img.Height -}} {{- else -}} {{- $src = $src | relURL -}} {{- end -}}
